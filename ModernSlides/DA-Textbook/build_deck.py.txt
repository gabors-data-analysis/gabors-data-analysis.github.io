def slide(id, blocks, speaker="", layout="auto", header="", aside="", print_="", steps=0, background=None):
    return {"id": id, "hidden": False, "layout": layout, "background": background, "header": header,
            "aside": aside, "speaker": speaker, "print": print_, "steps": steps, "blocks": blocks}


def title(t):
    return {"type": "title", "content": t, "step": 0}


def text(t, size="normal", step=0, align=None):
    b = {"type": "text", "size": size, "content": t, "step": step}
    if align:
        b["align"] = align
    return b


def ulist(items, size="normal", ordered=False, progressive=False):
    out = []
    for i, it in enumerate(items):
        d = {"content": it}
        if ordered:
            d["number"] = i + 1
        out.append(d)
    return {"type": "list", "ordered": ordered, "size": size, "items": out, "step": 0,
            "start": 1, "progressive": progressive}


def table(rows, align=None, widths=None):
    return {"type": "table", "widths": widths or [], "headerRows": 1, "headerCols": 0,
            "rows": rows, "step": 0, "columnAlign": align or []}


def cols(widths, *columns, valign="center"):
    return {"type": "columns", "widths": widths, "valign": valign,
            "columns": [{"blocks": c} for c in columns], "step": 0}


def image(src, caption="", width="100%"):
    return {"type": "image", "src": src, "caption": caption, "width": width, "fit": "contain",
            "focus": "center", "step": 0}


def source(t):
    return text(t, "tiny")
